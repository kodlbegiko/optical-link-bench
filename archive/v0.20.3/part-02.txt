function updateROI(loc,ox,oy,w,h){let xs=[loc.topLeftCorner.x,loc.topRightCorner.x,loc.bottomLeftCorner.x,loc.bottomRightCorner.x].map(x=>x+ox),ys=[loc.topLeftCorner.y,loc.topRightCorner.y,loc.bottomLeftCorner.y,loc.bottomRightCorner.y].map(y=>y+oy);let x0=Math.min(...xs),x1=Math.max(...xs),y0=Math.min(...ys),y1=Math.max(...ys),pad=Math.max(x1-x0,y1-y0)*.18;roi={x:clamp(x0-pad,0,w),y:clamp(y0-pad,0,h),w:clamp(x1-x0+2*pad,20,w),h:clamp(y1-y0+2*pad,20,h)};drawROI()}
